   792→const REVIEW_DAYS = [2, 5, 8, 12, 16, 20, 25, 32, 40, 50, 65, 80, 95, 110, 130, 160];
   793→
   794→export function generateReviews(productIndex: number, count: number): SeedReview[] {
   795→  // Deterministic pseudo-random based on product index so seed is reproducible
   796→  const reviews: SeedReview[] = [];
   797→  let seed = productIndex * 17 + 3;
   798→  const rand = () => {
   799→    seed = (seed * 9301 + 49297) % 233280;
   800→    return seed / 233280;
   801→  };
   802→  for (let i = 0; i < count; i++) {
   803→    const tpl = REVIEW_TEMPLATES[Math.floor(rand() * REVIEW_TEMPLATES.length)];
   804→    const author = REVIEW_AUTHORS[Math.floor(rand() * REVIEW_AUTHORS.length)];
   805→    const daysAgo = REVIEW_DAYS[Math.floor(rand() * REVIEW_DAYS.length)];
   806→    const helpful = Math.floor(rand() * 35);
   807→    const verified = rand() > 0.1;
   808→    reviews.push({
   809→      author,
   810→      rating: tpl.rating,
   811→      title: tpl.title,
   812→      body: tpl.body,
   813→      helpful,
   814→      verified,
   815→      daysAgo,
   816→    });
   817→  }
   818→  // Sort newest first (smaller daysAgo first)
   819→  return reviews.sort((a, b) => a.daysAgo - b.daysAgo);
   820→}
   821→
   822→// Number of reviews per product (varies between 3-8)
   823→export function reviewCountForProduct(productIndex: number): number {
   824→  return 3 + (productIndex % 6); // 3..8
   825→}
